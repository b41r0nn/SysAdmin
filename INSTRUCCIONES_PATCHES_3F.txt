# ARCHIVO 5 (OPCIONAL): backend/usuarios/views.py — detalle_usuario

# En la vista detalle_usuario, los activos asignados al usuario
# estaban retornando una lista vacía []. Ahora que inventario existe,
# reemplazar la vista por esta versión completa:

from inventario.models import Asignacion   # ← agregar import

@login_required
def detalle_usuario(request, pk):
    usuario = get_object_or_404(Usuario, pk=pk)
    # Activos con asignación activa
    activos = [
        a.activo for a in Asignacion.objects.filter(
            usuario=usuario, activa=True
        ).select_related("activo")
    ]
    context = {
        "usuario": usuario,
        "activos": activos,   # ← reemplaza la lista vacía []
    }
    return render(request, "usuarios/detalle.html", context)
